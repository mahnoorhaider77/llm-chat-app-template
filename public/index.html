import { getRelevantContext } from "./rag"; export default { async fetch(request: Request, env: any) { try { const { message } = await request.json(); // 1. Get RAG context from Vectorize const context = await getRelevantContext(env, message); // 2. Build prompt for AI const systemPrompt = ` You are a helpful AI assistant inside a chat app. Use the context below if it is relevant: CONTEXT: ${context || "No relevant context found."} Rules: - If context is relevant, prioritize it - If not, answer normally - Keep answers clear and simple `; // 3. Call Cloudflare AI (LLM) const aiResponse = await env.AI.run( "@cf/meta/llama-3-8b-instruct", { messages: [ { role: "system", content: systemPrompt }, { role: "user", content: message } ] } ); // 4. Return response in format your frontend can use return new Response( JSON.stringify({ response: aiResponse.response || aiResponse }), { headers: { "Content-Type": "application/json" } } ); } catch (err: any) { return new Response( JSON.stringify({ error: "Something went wrong", details: err.message }), { status: 500, headers: { "Content-Type": "application/json" } } ); } } };
